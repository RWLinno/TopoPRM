\section{Method}
\label{sec:method}

\begin{figure*}[!t]
\centering
\includegraphics[width=0.98\linewidth]{figures/fig2_topoprm_framework.pdf}
\caption{Overview of our proposed \method with post-training framework \framework. The DAG Scorer lifts reasoning traces into structural representations. A hierarchical multiplicative reward and \estimation bound within-stratum credit, while a topology-guided on-policy distillation stage compresses the long-CoT teacher into a compact student.}
\label{fig:overview}
\end{figure*}

\framework (Figure~\ref{fig:overview}) is a topology-aware post-training framework whose three stages share a \textbf{DAG Scorer} $\mathcal{E}$ that converts reasoning traces into structure-sensitive representations. Stage~I performs a supervised cold start to enforce parsable formatting. Stage~II runs GRPO with a hierarchical reward that fuses \method with \estimation. Stage~III executes topology-guided self-distillation, in which the Stage-II teacher provides token-level reverse-KL supervision while the student rolls out under topology-conditioned revision prompts.

\subsection{Problem Setup}
Let $x=(q,c)$ denote an input pairing a mathematical problem $q$ with optional context $c$, and let $\pi_\theta$ be a policy that generates a reasoning trace $y=(s_1,\dots,s_T)\sim\pi_\theta(\cdot\!\mid\!x)$, where each $s_i$ is a textual reasoning step. Outcome-only RLVR supervises $y$ with a sparse final-answer reward $r_{\mathrm{out}}(x,y)\!\in\![0,1]$. We define $\mathcal{G}_y=(\mathcal{V},\mathcal{A})$ as a directed dependency graph on $y$, with node set $\mathcal{V}=\{s_1,\dots,s_T\}$ and edge set $\mathcal{A}\subseteq\mathcal{V}\!\times\!\mathcal{V}$ encoding surface-evidenced support relations between steps, and $\mathcal{E}:y\!\mapsto\!\mathcal{G}_y$ as the DAG Scorer that produces it. \textit{Our objective is to recover latent dependency structure from $\mathcal{G}_y$ as unified supervision across SFT, RL, and distillation, without human step labels or learned verifier models.}

\subsection{Topology-Aware Data Processing}
\label{sec:data_processing}

\paragraph{Trace-to-DAG Extraction.}
Given a rollout $y=(s_1,\dots,s_T)$, the DAG Scorer $\mathcal{E}$ proceeds in three coupled stages whose outputs are reused by every downstream signal. \textbf{(i)~Step segmentation and parsing.} $\mathcal{E}$ first splits $y$ into reasoning steps using numbered markers, mathematical discourse markers, and line boundaries, then partitions the trace into sub-questions so that no edge can cross independent components. \textbf{(ii)~Implicit feature extraction.} Each step $s_i$ is reduced to three normalized feature sets: mathematical expressions $E_i$, canonicalized claim keys $C_i$, and variable references $V_i$, after information-density filters discard single symbols, pure numerals, and heading-like fragments that would otherwise inflate the matching surface. \textbf{(iii)~DAG construction.} For every ordered intra-component pair $(s_i,s_j)$ with $i\!<\!j$, $\mathcal{E}$ inserts an implicit edge $i\!\to\!j$ whenever $E_i\!\cap\!E_j$, $C_i\!\cap\!C_j$, or $V_i\!\cap\!V_j$ is non-empty; if $s_j$ is a derivation or conclusion step still lacking incoming support, an \emph{adaptive sequential weak} edge from $s_{j-1}$ is added only when $s_{j-1}$ does not already emit strong outgoing dependencies, which prevents the chain-dominant artefacts that long traces tend to produce. Full extraction rules are deferred to Appendix~\ref{app:dag}.

\paragraph{Topology and Continuity Diagnostics.}
\label{sec:diagnostics}
On the resulting graph $\mathcal{G}_y\!=\!(\mathcal{V},\mathcal{A})$ we compute two normalized scores that constitute the process-level reward signal. The \emph{topology quality} $q_{\mathrm{topo}}\!\in\![0,1]$ aggregates five structural checks, while the \emph{continuity quality} $q_{\mathrm{cont}}\!\in\![0,1]$ measures step-level traceability. The simplified formula is as follows:
\begin{align}
q_{\mathrm{topo}}(\mathcal{G}_y) &= \mathbb{1}[|\mathcal{V}|>0]
+\mathbb{1}[\mathcal{G}_y\text{ acyclic}] \notag\\
+\mathbb{1}[\rho_{\mathrm{orph}}&(\mathcal{G}_y)\!=\!0]+\delta(\mathcal{G}_y)
+\kappa(\mathcal{G}_y,\mathcal{G}^{\star}), \label{eq:qtopo}\\[2pt]
q_{\mathrm{cont}}(y) &= \gamma\,\eta(y)+(1-\gamma)\,\mathbb{1}[\eta(y)\!=\!1], \label{eq:qcont}\\[2pt]
r_{\mathrm{topo}}(y) &= \alpha\,q_{\mathrm{topo}}(\mathcal{G}_y)+(1-\alpha)\,q_{\mathrm{cont}}(y), \label{eq:rtopo}
\end{align}
where $\gamma,\alpha\!\in\!(0,1)$, $\rho_{\mathrm{orph}}$, $\delta$, and $\kappa$ measure, respectively, the fraction of orphan conclusion nodes, the proportion of edges respecting the premise-to-consequence ordering, and the alignment with an optional reference DAG $\mathcal{G}^{\star}$; $\eta(y)=\tfrac{1}{T}\sum_{i=1}^{T}\mathbb{1}[\mathrm{supported}(s_i,s_{<i},x)]$
is the fraction of reasoning steps in $y$ whose entities overlap with the prefix $s_{<i}$ or the question $x$. Detailed formulas and optimizations are reported in Appendix~\ref{app:hparam}.

\paragraph{Hierarchical Aggregation.}
Weighted-sum rewards can misrank traces when high process scores compensate for incorrect final answers~\citep{Denison2024SycophancyTS}, and they tend to collapse the within-group variance on which GRPO relies~\citep{yu2025dapo,kazemnejad2025vineppo}. We therefore couple the correctness-based and structural signals multiplicatively, treating $r_{\mathrm{topo}}$ as a structural modulator on top of an outcome-anchored base:
\begin{equation}
\label{eq:r_total}
\begin{aligned}
r_{\mathrm{total}}(x,y)=\;
&\bigl(w_o\,r_{\mathrm{out}}+w_f\,r_{\mathrm{fmt}}+w_l\,r_{\mathrm{len}}\bigr)\\
&\cdot \mathrm{Norm}_{\sigma}\! \bigl(1+r_{\mathrm{topo}}(y)\bigr),
\end{aligned}
\end{equation}
where $r_{\mathrm{out}}\!\in\![0,1]$ measures final-answer correctness, $r_{\mathrm{fmt}}$ scores parsability, and $r_{\mathrm{len}}$ regularises length. The structural modulator is wrapped by a batch-wise normalizer $\mathrm{Norm}_{\sigma}(\cdot)$, and the multiplicative form preserves correctness primacy: a trace with $r_{\mathrm{out}}\!=\!0$ cannot be lifted by any process bonus, while traces with equal outcome scores remain distinguishable through their dependency structure. This yields the within-group reward dispersion that \estimation exploits in Stage~II.

\subsection{Topology-Guided Post-Training}
\label{sec:post_training}
% The three stages of \framework share a single DAG-based supervision interface.
% Stage~I (SFT) aligns format and basic reasoning behaviour, Stage~II (GRPO with hierarchical reward) injects topology-sensitive process supervision without overriding answer correctness, and Stage~III (\framework) reuses the same diagnostics for compact-student distillation.

\paragraph{Stage I: Supervised Warm-Start.}
We perform Supervised Fine-Tuning (SFT) on structured mathematical reasoning data to serve as a warm-start that teaches the model to follow the required response format, produce stepwise solutions, and expose intermediate reasoning in a form parsable by $\mathcal{E}$. This reduces invalid rollouts during subsequent reinforcement learning and stabilizes reward computation, since topology diagnostics are meaningful only when the generated trace contains recoverable reasoning steps.

\paragraph{Stage II: GRPO with TopoPRM Rewards.}
Starting from the SFT checkpoint, we optimise the policy with GRPO under the \method reward. For each prompt, the policy samples a group of completions, each of which is parsed into a dependency DAG by $\mathcal{E}$ and scored accordingly. The aggregated reward in Eq.~\ref{eq:r_total} combines outcome correctness, format compliance, length regularization, and topology-derived process signals. Outcome rewards select for correct final answers, while topology and continuity rewards encourage the model to make intermediate dependencies explicit, avoid unsupported conclusions, and reduce structurally redundant reasoning.

Outcome rewards select for correct final answers, while topology and continuity rewards encourage the model to make intermediate dependencies explicit, avoid unsupported conclusions, and reduce structurally redundant reasoning. SCAE further constrains the auxiliary structural rewards so that they improve credit assignment within correctness strata rather than rewarding plausible-looking but wrong traces.

\paragraph{Asymmetric Clipping Advantage Estimator (\estimation).}
Multiplicative structural rewards introduce a potential hacking risk: a wrong response may appear well-structured under surface diagnostics. We propose ACE, an asymmetric clipping advantage estimator that fuses correctness-stratified credit assignment~\citep{liu2026graph} with standard GRPO z-score clipping~\citep{shao2024deepseekmath} into a single estimator,
\begin{equation}    
\label{eq:ace}
\widehat{A}_i =
\begin{cases}
\mathrm{clip}~\!\bigl(z_i^{\mathcal{C}} + \max(0,\Delta r_{\mathrm{aux}}^{(i)}),\, 0,\, \overline{c}\bigr), i\in\mathcal{C},\\[2pt]
\mathrm{clip}~\!\bigl(z_i^{\mathcal{W}} + \min(0,\Delta r_{\mathrm{aux}}^{(i)}),\, \underline{c},\, 0\bigr), i\in\mathcal{W},
\end{cases}
\end{equation}
where $\underline{c}<0<\overline{c}$ suppress tail noise within the assigned sign, $\mathcal{C}/\mathcal{W}$ are the correct/wrong strata defined by $r_{\text{out}}$, $z_i^{\pm}\!=\!(R_i-\mu_\pm)/(\sigma_\pm+\epsilon)$ is the within-stratum z-score, and $\Delta r_{\mathrm{aux}}^{(i)}$ is the deviation of $r_{topo}$ from its stratum mean. The asymmetric $\max/\min$ ensures structural signals can only re-rank traces \emph{within} a correctness stratum, never flip its sign; the bounds $(\underline{c},\overline{c})$ suppress tail noise. Together with batch-relative normalisation and a positive base floor for low-variance groups, this prevents reward collapse. Full derivation and the stable pseudo-code are deferred to Appendix~\ref{app:ace}.

\paragraph{Stage~III: Topology-Guided Self Distillation.}
\label{sec:tgsd}
The optimized model after Stage~II serves as the teacher $\pi_T$, while a smaller pretrained checkpoint initializes the student $\pi_\theta$. \framework performs on-policy distillation conditioned on topology diagnostics. For each prompt $x$, the student first rolls out its own response $y^{\star}\!\sim\!\pi_\theta(\cdot\!\mid\!x)$; the same extractor $\mathcal{E}$ produces $\mathcal{G}_{y^{\star}}$ together with $(r_{\mathrm{out}},q_{\mathrm{topo}},q_{\mathrm{cont}})$; a dispatcher $\psi$ then issues a revision instruction $P_r=\psi(r_{\mathrm{out}},q_{\mathrm{topo}},k)$ targeting an orphan-conclusion position $k$ in $\mathcal{G}_{y^{\star}}$. The teacher then supervises the student under $P_r$, so that the signal concentrates on structurally suspect regions (orphan conclusions and continuity breaks) rather than spreading uniformly across the trace. Sampling $y^{\star}$ from $\pi_\theta$ rather than from a fixed teacher trace eliminates the train--test mismatch of off-policy distillation~\citep{agarwal2024gkd,song2026opdsurvey}.


\begin{algorithm}[t]
\small
\caption{\framework Training Pipeline}
\label{alg:topoprm}
\begin{algorithmic}[1]
\Require Base policy $\pi_0$; data $\mathcal{D}_{\mathrm{sft}},\mathcal{D}_{\mathrm{rl}}$; weights $w_o,w_f,w_l,\alpha$
\Ensure Teacher $\pi_T$, compact student $\pi_S$
\State $\pi_\theta\!\leftarrow\!\mathrm{SFT}(\pi_0,\mathcal{D}_{\mathrm{sft}})$ \Comment{Stage~I}
\For{each GRPO step on $\mathcal{D}_{\mathrm{rl}}$} \Comment{Stage~II}
  \State Sample $\{y^{(i)}\}_{i=1}^{G}\!\sim\!\pi_\theta(\cdot\!\mid\!x)$; build $\mathcal{G}^{(i)}\!=\!\mathcal{E}(y^{(i)})$
  \State Compute $r_{\mathrm{out}}^{(i)},r_{\mathrm{fmt}}^{(i)},r_{\mathrm{len}}^{(i)},q_{\mathrm{topo}}^{(i)},q_{\mathrm{cont}}^{(i)}$
  \State $r_{\mathrm{total}}^{(i)}$ by Eq.~\ref{eq:r_total}; $\widehat{A}^{(i)}$ by Eq.~\ref{eq:ace}
  \State Update $\pi_\theta'$ with the GRPO objective
\EndFor
\State $\pi_T\!\leftarrow\!\pi_\theta'$; init.\ $\pi_S$ from a smaller pretrained checkpoint
\For{each prompt $x\!\in\!\mathcal{D}_{\mathrm{rl}}$} \Comment{Stage~III}
  \State $y_{\mathrm{init}}\!\sim\!\pi_S(\cdot\!\mid\!x)$; compute $(r_{\mathrm{out}},q_{\mathrm{topo}},q_{\mathrm{cont}})$ on $\mathcal{E}(y_{\mathrm{init}})$
  \State $P_r\!\leftarrow\!\psi(r_{\mathrm{out}},q_{\mathrm{topo}},k)$
  \State $y_{\mathrm{rev}}\!\sim\!\pi_T(\cdot\!\mid\!x,y_{\mathrm{init}},P_r)$; filter by format/topology
  \State Update $\pi_S$ with $\mathcal{L}_{\mathrm{TGSD}}$ (Eq.~\ref{eq:tgsd})
\EndFor
\State \Return $(\pi_T,\pi_S)$
\end{algorithmic}
\end{algorithm}

\paragraph{Distillation objective.}
Let $p_t \!\triangleq\! \pi_\theta(\cdot\!\mid\!x,y_{<t})$ and $q_t \!\triangleq\! \pi_T(\cdot\!\mid\!x,y^{\star},P_r,y_{<t})$ denote the student and teacher next-token distributions at position $t$, where $y_{<t}$ is the autoregressive prefix sampled on-policy by the student. Restricted to samples passing format and topology filters ($r_{\mathrm{out}}^{(\mathrm{rev})}\!=\!1$ and $q_{\mathrm{topo}}^{(\mathrm{rev})}\!\ge\!\tau_g$), the student minimises
\begin{equation}
\label{eq:tgsd}
\mathcal{L}_{\mathrm{TGSD}}(\theta)=
\mathbb{E}_{x,\,y\sim\pi_\theta}\!\sum_{t=1}^{|y|}
\mathcal{D}_{\mathrm{RKL}}\!\bigl(p_t\,\|\,q_t\bigr).
\end{equation}
$\mathcal{G}_{y^{\star}}$ exposes broken support and $P_r$ steers $\pi_T$ toward the missing dependency, turning the per-token RKL into a structurally targeted gradient.

\paragraph{Critical dependency chain as the compression target.}
Equation~\eqref{eq:tgsd} pushes the student toward a linearisation of $\mathcal{G}_{y^{\star}}$ obtained by composing three operations on the DAG: \emph{step-type contraction} merges consecutive same-type nodes; \emph{layer aggregation} folds parallel topological generations into a layered chain $L_1\!\to\!L_2\!\to\!\cdots$; and \emph{transitive sparsification} replaces $\mathcal{G}$ by its transitive reduction $\mathrm{tr}(\mathcal{G})$. The resulting \emph{critical dependency chain}, analogous to the critical path in project scheduling, lower-bounds any structurally supported trajectory that closes the answer; the student is therefore trained to recover this minimal chain rather than to match teacher token frequencies, which is the structural counterpart of length-penalty compression. The full pipeline is summarised in Algorithm~\ref{alg:topoprm}.
